```tsx
export function CardFace({ card }: { card: Card }) {

  );
  return (
    <>
      {idx(false)}
      <span className="card__pip" aria-hidden="true">{s}</span>

}
```

```ts
/** Minimum hand-card width per project (px). */
const MIN_W: Record<string, number> = { se: 70, iphone15: 76, promax: 84, landscape: 38, desktop: 90 };

async function handMetrics(page: Page) {
  return page.locator('.hand .hand__card').evaluateAll(els =>
    els.map(el => {
      const r = el.getBoundingClientRect();
      const rank = el.querySelector('.card__idx .card__rank')!.getBoundingClientRect();
      return { left: r.left, right: r.right, width: r.width, rankRight: rank.right, rankH: rank.height };
    }),
  );
}

test('a 10-card hand is big, fully on screen and every index is readable', async ({ page }, info) => {
  await firstLaunch(page);
  await page.getByRole('button', { name: 'Practice vs bots' }).click();
  await expect(page.locator('.zone')).toBeVisible({ timeout: 20_000 });
  // Win the auction so the hand holds 10 cards (decision phase, before the hand-out).
  await drive(page, {
    bidAlways: true,
    check: async () => {},
    stop: async () => (await page.locator('.hand .hand__card').count()) === 10,

  const cards = await handMetrics(page);
  const vw = page.viewportSize()!.width;
  expect(cards).toHaveLength(10);
```

```ts
/** [foreground, background] token pairs that carry numbers or card ranks. */
```

```ts
import { SUIT_SYMBOL, cardLabel, type Card, type GameEvent, type PlayerId, type PlayerView, type Suit } from '@tysiac/engine';
import { who } from '../store/notes';

// (Announcement type exactly as in Interfaces above.)

export function announcementFor(e: GameEvent, me: PlayerId | null): Announcement | null {
  switch (e.type) {
    case 'dealt': return { kind: 'deal', round: e.round, dealer: e.dealer };
    case 'bid': return { kind: 'bid', p: e.p, amount: e.amount };
    case 'passed': return { kind: 'pass', p: e.p };
    case 'biddingWon': return { kind: 'bidWon', p: e.p, amount: e.amount };
    case 'threw': return { kind: 'threw', p: e.p, amount: e.amount };
    case 'handedOut': {
      const mine = me !== null && e.p !== me ? e.cards[me] : undefined;
      return mine ? { kind: 'received', from: e.p, card: mine } : { kind: 'handout', p: e.p };
    }
    case 'wykladana': return { kind: 'wykladana', p: e.p };
    case 'cardPlayed': return e.p === me ? null : { kind: 'played', p: e.p, card: e.card };
    case 'marriage': return { kind: 'marriage', p: e.p, suit: e.suit, points: e.points };
    case 'trickWon': return { kind: 'trick', p: e.p, points: e.points };
    case 'gameWon': return { kind: 'won', p: e.p };
    case 'paused': return { kind: 'paused', p: e.p };
    case 'resumed': return { kind: 'resumed', p: e.p };
    default: return null;
  }
}

const isMe = (v: PlayerView, p: PlayerId) => p === v.me;

/** English text. Stream P replaces every literal here with t(...). */
export function announceText(a: Announcement, v: PlayerView): string {
  switch (a.kind) {
    case 'deal': return `Round ${a.round} — ${isMe(v, a.dealer) ? 'you deal' : `${who(v, a.dealer)} deals`}`;
    case 'bid': return `${who(v, a.p)} bid ${a.amount}`;
    case 'pass': return `${who(v, a.p)} passed`;
    case 'bidWon': return `${who(v, a.p)} won the bid at ${a.amount}`;
    case 'threw': return `${who(v, a.p)} threw in at ${a.amount}`;
    case 'handout': return isMe(v, a.p) ? 'You handed out 2 cards' : `${who(v, a.p)} handed out cards`;
    case 'received': return `You got ${cardLabel(a.card)} from ${who(v, a.from)}`;
    case 'wykladana': return `${who(v, a.p)} won every trick — wykładana!`;
    case 'played': return `${who(v, a.p)} played ${cardLabel(a.card)}`;
    case 'marriage': {
      const s = SUIT_SYMBOL[a.suit];
      return `${who(v, a.p)}: marriage ${s} +${a.points} — trump is ${s}`;
    }
    case 'trick': return `${who(v, a.p)} ${isMe(v, a.p) ? 'take' : 'takes'} the trick (+${a.points})`;
    case 'won': return `${who(v, a.p)} ${isMe(v, a.p) ? 'win' : 'wins'} the game!`;
    case 'paused': return `${who(v, a.p)} paused the game`;
    case 'resumed': return `${who(v, a.p)} resumed the game`;
  }
}
```

```ts
interface Synth { type: OscillatorType; vol: number; notes: Note[] }
const seq = (fs: number[], dur: number): Note[] => fs.map((f, i) => ({ f: [f], at: i * dur * 0.6, dur }));

const SYNTH: Record<SoundName, Synth> = {
  card: { type: 'triangle', vol: 0.18, notes: seq([180], 0.06) },
  shuffle: { type: 'sawtooth', vol: 0.05, notes: seq([140, 190, 160], 0.12) },
  sweep: { type: 'triangle', vol: 0.08, notes: seq([320, 240], 0.14) },
  turn: { type: 'sine', vol: 0.1, notes: seq([660, 880], 0.18) },
  // Old-app recipes (packages/game-logic SOUND_SPECS).
  marriage: { type: 'sine', vol: 0.08, notes: [{ f: [523, 659, 784], at: 0, dur: 0.4 }, { f: [587, 740, 880], at: 0.3, dur: 0.5 }] },
  roundEnd: { type: 'triangle', vol: 0.08, notes: [{ f: [440, 554, 659], at: 0, dur: 0.6 }] },
  gameWin: {
    type: 'sine', vol: 0.1,
    notes: [...seq([523, 659, 784, 1047], 0.25).map((n, i) => ({ ...n, at: i * 0.15 })), { f: [1047, 1319, 1568], at: 0.6, dur: 0.8 }],
  },
};

  private synth(ctx: AudioContextLike, s: Synth): void {
    for (const n of s.notes) {
      for (const f of n.f) {
        const osc = ctx.createOscillator();
        const gain = ctx.createGain();
        const t = ctx.currentTime + n.at;
        osc.type = s.type;
        osc.frequency.value = f;
        gain.gain.setValueAtTime(s.vol / Math.sqrt(n.f.length), t);
        gain.gain.exponentialRampToValueAtTime(0.001, t + n.dur);
        osc.connect(gain);
        gain.connect(ctx.destination);
        osc.start(t);
        osc.stop(t + n.dur);
      }
    }
  }
```

```ts
/**
 * The player a bot must move for right now, or null (paused, finished, only humans to act, or nothing
 * for the engine bot to do). In the talon phase several seats may act; the first bot-controlled one goes.
 */
export function botActor(state: GameState, meta: RoomMeta): PlayerId | null {
  if (state.paused !== null || state.winner !== null || state.phase === 'gameEnd') return null;
  for (const actor of actorsOf(state)) {
    if (isBotControlled(meta, actor) && botShouldAct(viewFor(state, actor))) return actor;
  }
  return null;
}
```

# Stream P — Polish and Danish (sequential, after I1)

## Glossary (binding for every P task)

Style rules:
- **Polish:** avoid gendered past tense for players (we don't know their gender). Use present tense or noun phrases: "Basia licytuje 120", "Licytację wygrywa Basia (140)", "Lewę bierze Zenek (+25)". Use informal "ty" (Twój ruch). Plurals have three forms: `one` (1), `few` (n%10 ∈ 2–4 and n%100 ∉ 12–14), `many` (everything else, including 0, 5–21, 25–31…). Points: "1 punkt / 2 punkty / 5 punktów", short form "pkt".
- **Danish:** informal "du". Plurals `one`/`other`. Points: "1 point / 5 point".
- A sentence whose subject is the viewer gets its own key with the suffix `.you` (verb forms differ: "You take" / "Bierzesz" / "Du tager"), so `{player}` never stands for "You" as a subject.
- Card rank letters on faces and in text: en `A K Q J 10 9`; pl `A K D W 10 9`; da `E K D B 10 9` (Es, Konge, Dame, Bonde/knægt). Suit glyphs never change.
- Keep "Tysiąc" untranslated everywhere (title, manifest, Home). Theme names (Brass, Wycinanki, Midnight) stay as they are.

`i18n/index.ts`:

```ts
import { en, type Dict, type MsgKey } from './en';
import type { Lang, Msg, Params } from './types';

let dict: Dict = en;
let lang: Lang = 'en';
const loaders: Record<Exclude<Lang, 'en'>, () => Promise<Dict>> = {
  pl: () => import('./pl').then(m => m.pl),
  da: () => import('./da').then(m => m.da),
};

export function currentLang(): Lang {
  return lang;
}

/** Swaps the active dictionary. A failed chunk load (offline, not cached) keeps the current one and rethrows. */
export async function loadLang(next: Lang): Promise<void> {
  dict = next === 'en' ? en : await loaders[next]();
  lang = next;
  if (typeof document !== 'undefined') document.documentElement.lang = next;
}

export function pluralCategory(l: Lang, n: number): 'one' | 'few' | 'many' | 'other' {
  const a = Math.abs(n);
  if (l !== 'pl') return a === 1 ? 'one' : 'other';
  if (a === 1) return 'one';
  const d = a % 10;
  const h = a % 100;
  return d >= 2 && d <= 4 && (h < 12 || h > 14) ? 'few' : 'many';
}

function pick(m: Msg, params?: Params): string {
  if (typeof m === 'string') return m;
  const n = Number(params?.n ?? 0);
  const cat = pluralCategory(lang, n);
  return (cat === 'few' ? m.few : cat === 'many' ? m.many : cat === 'one' ? m.one : undefined) ?? m.other;
}

export function t(key: MsgKey, params?: Params): string {
  const text = pick(dict[key] ?? en[key], params);
  return params ? text.replace(/\{(\w+)\}/g, (all, k: string) => (k in params ? String(params[k]) : all)) : text;
}
```

`engineText.ts`:

```ts
import { SUIT_SYMBOL, type Card, type EngineMsg, type MsgParam, type PlayerId, type PlayerView } from '@tysiac/engine';
import type { MsgKey } from './en';
import { currentLang, t } from './index';

export function cardText(card: Card): string {
  return `${t(`card.rank.${card.rank}` as MsgKey)}${SUIT_SYMBOL[card.suit]}`;
}
export function playerName(view: PlayerView | null, p: PlayerId): string {
  if (view && p === view.me) return t('common.you');
  return view?.seats.find(s => s.id === p)?.name ?? t('common.someone');
}
function param(v: MsgParam, view: PlayerView | null): string | number {
  if (typeof v === 'string' || typeof v === 'number') return v;
  if ('card' in v) return cardText(v.card);
  if ('suit' in v) return SUIT_SYMBOL[v.suit];
  if ('player' in v) return playerName(view, v.player);
  if ('signed' in v) return v.signed > 0 ? `+${v.signed}` : v.signed < 0 ? `−${-v.signed}` : '0';
  if ('list' in v) return new Intl.ListFormat(currentLang(), { type: 'conjunction' }).format(v.list.map(x => String(param(x, view))));
  return v.msgs.map(m => engineText(m, view)).join(' ');
}
export function engineText(m: EngineMsg, view: PlayerView | null): string {
  const params = Object.fromEntries(Object.entries(m.params ?? {}).map(([k, v]) => [k, param(v, view)]));
  return t(`engine.${m.key}` as MsgKey, params);
}
```
