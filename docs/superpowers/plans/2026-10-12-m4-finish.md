> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Finish the rebuild. This adds a scripted "Learn to play" round for new players, a History screen fed by `GET /api/history`, Playwright e2e against the real Worker under `wrangler dev`, a CI workflow, new worker debugging notes in CLAUDE.md, and a cutover checklist that only the owner can carry out.

**Architecture:** The lesson is one deterministic round. `buildLesson(LESSON_SEED)` simulates it with the engine (the learner follows the engine's `advise`, except that it leads the marriage Queen on trick 1; bots follow `botAction`) and records every action with its `seq`. At runtime an in-browser `LocalRoom` with the same seed replays the bots' moves from that transcript (new `botMove` option). A `LessonConnection` wraps the `LocalConnection` and turns any learner action that is not the scripted one into a friendly `reject`. A `gatedClock` holds bot timers while an explanation sheet is open. The real `Game` screen renders the lesson unchanged except for one context hook, which lets the lesson supply the coach line (`Advice`). Explanations are a list of "beats" that read the engine `RULES` and the player's view. History is a lazy screen with an internal scroll list. Worker e2e runs `wrangler dev` serving the built app, with two browser contexts.

**Tech Stack:** TypeScript 5.9 strict, React 19.3, zustand 5, wouter 3, Vite 8, vitest 4.1.11 (root override, do not bump), Playwright 1.64 (WebKit), wrangler 4, GitHub Actions on Node 24, and pnpm 9.

import {
  RULES, advise, botAction, cardEquals, createGame, makeRandom, marriagesInHand, reduce, turnOf, viewFor,
  type Action, type Card, type PlayerId, type PlayerView, type RoundSummary, type Seat, type Suit,
} from '@tysiac/engine';
import { BOT_AVATAR, BOT_NAMES } from '../net/localRoom';

/** The learner's id inside the lesson room (local only; never sent to a server). */
export const LEARNER_ID: PlayerId = 'learner';
export const LESSON_SECRET = 'lesson-local-secret';
/** LocalRoom names its first two bots `bot-1`/`bot-2` and "Bot Basia"/"Bot Zenek". */
export const LESSON_BOT_IDS = ['bot-1', 'bot-2'] as const;
export const LESSON_FIRST_DEALER = 0;
/** Found with findSeed.test.ts; lesson.test.ts proves it still teaches every beat. */
export const LESSON_SEED = 'lesson-0';

export interface LessonEntry {
  seq: number;
  pid: PlayerId;
  action: Action;
}
export interface LessonMoments {
  marriage: Suit | null;
  learnerBid: boolean;
  bidWon: number | null;
  played: boolean;
  marriageLead: boolean;
  mustBeat: boolean;
  noSuit: boolean;
  wykladana: boolean;
}
export interface Lesson {
  seed: string;
  entries: LessonEntry[];
  summary: RoundSummary | null;
  moments: LessonMoments;
}

export function lessonSeats(name = 'You', avatar = '🙂'): Seat[] {
  return [
    { id: LEARNER_ID, name, avatar, isBot: false },
    { id: LESSON_BOT_IDS[0], name: BOT_NAMES[0], avatar: BOT_AVATAR, isBot: true },
    { id: LESSON_BOT_IDS[1], name: BOT_NAMES[1], avatar: BOT_AVATAR, isBot: true },
  ];
}

const legalCards = (v: PlayerView): Card[] => v.legal.flatMap(a => (a.type === 'card' ? [a.card] : []));
const myTricksTurn = (v: PlayerView): boolean => v.phase === 'tricks' && v.turn === v.me && !v.paused && v.round !== null;

/** On trick 1, the learner's lead of the Queen of an undeclared marriage (if legal), else null. */
export function marriageLead(v: PlayerView): Action | null {
  if (!myTricksTurn(v)) return null;
  const r = v.round!;
  if (r.trick.cards.length !== 0 || r.played.length !== 0) return null;
  const declared = r.marriages[v.me] ?? [];
  const suit = marriagesInHand(r.hand).find(s => !declared.includes(s));
  if (!suit) return null;
  const queen = legalCards(v).find(c => c.suit === suit && c.rank === 'Q');
  return queen ? { type: 'card', card: queen } : null;
}

/** First to answer, and the engine forbids some card of the led suit, so the must-beat rule is in force. */
export function mustBeatNow(v: PlayerView): boolean {
  if (!myTricksTurn(v) || v.round!.trick.cards.length !== 1) return false;
  const led = v.round!.trick.cards[0].card.suit;
  const legal = legalCards(v);
  return v.round!.hand.some(c => c.suit === led && !legal.some(l => cardEquals(l, c)));
}

/** The learner must answer a trick and holds no card of the led suit. */
export function cannotFollowNow(v: PlayerView): boolean {
  if (!myTricksTurn(v) || v.round!.trick.cards.length === 0) return false;
  const led = v.round!.trick.cards[0].card.suit;
  return !v.round!.hand.some(c => c.suit === led);
}

/**
 * Plays round 1 of the lesson game: bots use botAction (own seeded rng), the learner follows advise(),
 * except it leads the marriage Queen on trick 1. Stops at the round's `roundScored`/`threw`.
 */
export function buildLesson(seed: string): Lesson {
  let state = createGame({ seed: `${seed}:0`, seats: lessonSeats(), maxPlayers: 3, firstDealer: LESSON_FIRST_DEALER });
  const rng = makeRandom(`${seed}:lesson-bots`);
  const entries: LessonEntry[] = [];
  const dealt = viewFor(state, LEARNER_ID).round?.hand ?? [];
  const moments: LessonMoments = {
    marriage: marriagesInHand(dealt)[0] ?? null,
    learnerBid: false, bidWon: null, played: false, marriageLead: false, mustBeat: false, noSuit: false, wykladana: false,
  };
  let summary: RoundSummary | null = null;
  for (let guard = 0; guard < 300; guard++) {
    if (summary !== null) break;
    const pid = turnOf(state);
    if (!pid) break;
    const view = viewFor(state, pid);
    let action: Action;
    if (pid === LEARNER_ID) {
      if (mustBeatNow(view)) moments.mustBeat = true;
      if (cannotFollowNow(view)) moments.noSuit = true;
      const lead = marriageLead(view);
      if (lead) moments.marriageLead = true;
      action = lead ?? advise(view)?.action ?? botAction(view, rng);
      if (action.type === 'bid') moments.learnerBid = true;
      if (action.type === 'play') moments.played = true;
    } else {
      action = botAction(view, rng);
    }
    const r = reduce(state, pid, action, 0);
    if (!r.ok) throw new Error(`lesson ${seed}: ${pid} ${JSON.stringify(action)} rejected: ${r.error.message}`);
    entries.push({ seq: state.seq, pid, action });
    for (const e of r.events) {
      if (e.type === 'wykladana') moments.wykladana = true;
      if (e.type === 'biddingWon' && e.p === LEARNER_ID) moments.bidWon = e.amount;
      if (e.type === 'roundScored' || e.type === 'threw') summary = e.summary;
    }
    state = r.state;
  }
  return { seed, entries, summary, moments };
}

/** Every beat the lesson promises (spec §9) that this seed fails to produce. Empty = usable seed. */
export function lessonProblems(l: Lesson): string[] {
  const m = l.moments;
  const out: string[] = [];
  if (!m.marriage) out.push('learner is dealt no marriage');
  if (!m.learnerBid) out.push('learner never bids');
  if (m.bidWon === null || m.bidWon <= RULES.autoBid) out.push('learner does not win the bidding above 100');
  if (!m.played) out.push('learner does not choose Play');
  if (m.wykladana) out.push('round ends as wykładana');
  if (!m.marriageLead) out.push('learner does not lead a marriage Queen on trick 1');
  if (!m.mustBeat) out.push('no must-beat moment for the learner');
  if (!m.noSuit) out.push('no cannot-follow moment for the learner');
  if (!l.summary || l.summary.bidder !== LEARNER_ID || l.summary.outcome !== 'made') out.push('learner does not make the bid');
  return out;
}

export function expectedAt(l: Lesson, seq: number): Action | null {
  const e = l.entries.find(x => x.seq === seq);
  return e && e.pid === LEARNER_ID ? e.action : null;
}

export function lessonBotMove(l: Lesson): (view: PlayerView, pid: PlayerId) => Action | null {
  return (view, pid) => {
    const e = l.entries.find(x => x.seq === view.seq);
    return e && e.pid === pid ? e.action : null;
  };
}

import { realClock, type Clock } from '../net/localRoom';

/** Open = timers run when due; closed = due timers wait until the gate opens. */
export class Gate {
  private open = true;
  private readonly waiters = new Set<() => void>();
  isOpen(): boolean {
    return this.open;
  }
  set(open: boolean): void {
    if (open === this.open) return;
    this.open = open;
    if (!open) return;
    const due = [...this.waiters];
    this.waiters.clear();
    for (const fn of due) fn();
  }
  /** Runs fn now if open, else when the gate next opens. Returns a cancel function. */
  whenOpen(fn: () => void): () => void {
    if (this.open) {
      fn();
      return () => {};
    }
    this.waiters.add(fn);
    return () => {
      this.waiters.delete(fn);
    };
  }
}

/** A Clock whose timers hold while `gate` is closed (bots wait while the lesson explains). */
export function gatedClock(gate: Gate, base: Clock = realClock): Clock {
  return {
    now: () => base.now(),
    setTimeout(fn, ms) {
      let cancelWait = () => {};
      const cancelTimer = base.setTimeout(() => {
        cancelWait = gate.whenOpen(fn);
      }, ms);
      return () => {
        cancelTimer();
        cancelWait();
      };
    },
  };
}

import { actionsEqual, type Action, type PlayerView } from '@tysiac/engine';
import type { ClientMessage, ServerMessage } from '../net/protocol';
import { Emitter, type CloseReason, type ConnStatus, type RoomConnection } from '../net/connection';

export interface LessonRules {
  /** The learner's scripted action at this server seq, or null. */
  expectedAt(seq: number): Action | null;
  /** The friendly line shown when the learner tries something else. */
  hint(view: PlayerView | null, expected: Action | null, tried: Action): string;
}

/** Forwards everything except engine actions that are not the lesson's next move; those get a local reject. */
export class LessonConnection implements RoomConnection {
  private seq = -1;
  private view: PlayerView | null = null;
  private readonly msgs = new Emitter<ServerMessage>();
  private readonly offInner: () => void;

  constructor(private readonly inner: RoomConnection, private readonly rules: LessonRules) {
    this.offInner = inner.onMessage(m => {
      if (m.t === 'state') {
        this.seq = m.seq;
        this.view = m.view;
      }
      this.msgs.emit(m);
    });
  }

  get code(): string {
    return this.inner.code;
  }
  get status(): ConnStatus {
    return this.inner.status;
  }
  get closeReason(): CloseReason | undefined {
    return this.inner.closeReason;
  }

  send(msg: ClientMessage): boolean {
    if (msg.t !== 'act') return this.inner.send(msg);
    if (this.inner.status !== 'open') return false;
    const expected = this.rules.expectedAt(this.seq);
    if (expected && actionsEqual(expected, msg.action)) return this.inner.send(msg);
    const message = this.rules.hint(this.view, expected, msg.action);
    setTimeout(() => this.msgs.emit({ t: 'reject', id: msg.id, code: 'lesson', message }), 0);
    return true;
  }

  onMessage(fn: (msg: ServerMessage) => void): () => void {
    return this.msgs.on(fn);
  }
  onStatus(fn: (status: ConnStatus) => void): () => void {
    return this.inner.onStatus(fn);
  }
  close(): void {
    this.offInner();
    this.inner.close();
  }
}

async function perform(page: Page, e: ExpectedHint): Promise<void> {
  const hand = page.getByRole('group', { name: 'Your hand' });

    case 'handout':

/** Walks the whole lesson; calls check(label) at each beat and each move. Returns the beat ids seen. */
async function walkLesson(page: Page, check: (label: string) => Promise<void>): Promise<string[]> {
  const hook = page.getByTestId('lesson');
  const beats: string[] = [];
  let triedOffScript = false;
  for (let i = 0; i < 4000; i++) {
    if (await page.getByRole('button', { name: 'Practice with coach' }).isVisible()) {
      beats.push('done');
      await check('beat done');
      return beats;
    }
    const next = page.getByRole('button', { name: 'Next' });
    if (await next.isVisible()) {
      const id = (await hook.getAttribute('data-beat')) ?? '?';
      if (beats.at(-1) !== id) beats.push(id);
      await check(`beat ${id}`);
      await next.click();
      continue;
    }
    const cont = page.getByRole('button', { name: 'Continue' });
    if (await cont.isVisible()) {
      await cont.click();
      continue;
    }
    const raw = await hook.getAttribute('data-expected');
    if (raw) {
      const e = JSON.parse(raw) as ExpectedHint;
      await check(`move ${e.type}`);
      if (!triedOffScript && e.type === 'bid') {
        triedOffScript = true;
        await page.getByRole('button', { name: 'Pass' }).click();
        await expect(page.getByText(/^Lesson: Bid \d+\.$/)).toBeVisible();
      }
      await move(page, () => perform(page, e));
      continue;
    }
    await page.waitForTimeout(40);
  }
  throw new Error(`lesson did not finish; beats so far: ${beats.join(', ')}`);
}

  return BEATS.find(b => !seen.includes(b.id) && b.ready(v)) ?? null;

const PLAN = "That's the plan for this hand — later the coach will help you choose.";

/** The coach line during the lesson: the scripted move, explained (reasons from advise() when it agrees). */
export function lessonAdvice(v: PlayerView, expected: Action | null): Advice | null {
  if (!expected || v.turn !== v.me || !v.round) return null;
  const coach = advise(v);
  const theirs = coach && actionsEqual(coach.action, expected) ? coach.reasons : [];
  const why = (fallback: string): string[] => (theirs.length > 0 ? theirs : [fallback]);
  switch (expected.type) {
    case 'bid':
      return { action: expected, headline: `Bid ${expected.amount}`, reasons: why(`Your marriage lets you bid up to ${v.round.maxBid}.`) };
    case 'pass':
      return { action: expected, headline: 'Pass', reasons: why(PLAN) };
    case 'play':
      return { action: expected, headline: 'Play the hand', reasons: why(throwCost(v)) };
    case 'throw':
      return { action: expected, headline: 'Throw in', reasons: why(throwCost(v)) };
    case 'handout': {
      const parts = Object.entries(expected.to).map(([p, c]) => `${cardLabel(c)} to ${nameOf(v, p)}`);
      return { action: expected, headline: `Give ${parts.join(' and ')}`, reasons: why(PLAN) };
    }
    case 'card': {
      const lead = marriageLead(v);
      if (lead && actionsEqual(lead, expected)) {
        return {
          action: expected,
          headline: `Lead ${cardLabel(expected.card)} to declare your marriage`,
          reasons: [`+${RULES.marriageValues[expected.card.suit]} for you, and ${sym(expected.card.suit)} becomes trump.`],
        };
      }
      return { action: expected, headline: `Play ${cardLabel(expected.card)}`, reasons: why(PLAN) };
    }
    default:
      return null;
  }
}

export function lessonHint(v: PlayerView | null, expected: Action | null, tried: Action): string {
  if (tried.type === 'pause' || tried.type === 'resume') return 'The lesson waits for you — no need to pause.';
  if (!v || !expected) return 'Not yet — wait for your turn.';
  const a = lessonAdvice(v, expected);
  return a ? `Lesson: ${a.headline}.` : 'Not yet — wait for your turn.';
}

export interface ExpectedHint {
  type: Action['type'];
  amount?: number;
  card?: string;
  give?: { card: string; to: string }[];
}

export function describeExpected(v: PlayerView, a: Action): ExpectedHint {
  switch (a.type) {
    case 'bid':
      return { type: 'bid', amount: a.amount };
    case 'card':
      return { type: 'card', card: cardLabel(a.card) };
    case 'handout':
      return { type: 'handout', give: Object.entries(a.to).map(([p, c]) => ({ card: cardLabel(c), to: nameOf(v, p) })) };
    default:
      return { type: a.type };
  }
}

import type { HistoryResponse } from './protocol';
import { RoomApiError } from './httpService';

function isHistory(b: unknown): b is HistoryResponse {
  const o = b as Partial<HistoryResponse> | null;
  return !!o && typeof o === 'object' && Array.isArray(o.games) && Array.isArray(o.headToHead) && Array.isArray(o.leaderboard);
}

/** GET /api/history?player=<id> (spec §8). Errors are RoomApiError with a message safe to show. */
export async function fetchHistory(playerId: string, base = ''): Promise<HistoryResponse> {
  let res: Response;
  try {
    res = await fetch(`${base}/api/history?player=${encodeURIComponent(playerId)}`);
  } catch {
    throw new RoomApiError('Cannot reach the server. Check your connection.', 'network');
  }
  if (!res.ok) throw new RoomApiError(`Server error (HTTP ${res.status})`, 'server');
  let body: unknown;
  try {
    body = await res.json();
  } catch {
    body = undefined;
  }
  if (!isHistory(body)) throw new RoomApiError('Unexpected response from the server.', 'badResponse');
  return body;
}

import { readFileSync } from 'node:fs';
import { fileURLToPath } from 'node:url';
import { expect, test, type Page } from '@playwright/test';
import { drive } from '../e2e/helpers';

const DEV_VARS = fileURLToPath(new URL('../../worker/.dev.vars', import.meta.url));
const debugKey = (): string => /^DEBUG_KEY=(.+)$/m.exec(readFileSync(DEV_VARS, 'utf8'))![1].trim().replace(/^"|"$/g, '');

async function launch(page: Page, name: string): Promise<void> {
  await page.goto('/?local=0');
  await page.getByLabel('Your name').fill(name);
  await page.getByRole('button', { name: 'Avatar 🐻' }).click();
  await page.getByRole('button', { name: 'Continue' }).click();
  await expect(page.getByRole('button', { name: 'Practice vs bots' })).toBeVisible();
}

const afterFirstTrick = (p: Page) => async () =>
  (await p.getByLabel('Last trick').isVisible()) || (await p.getByRole('button', { name: 'Continue' }).isVisible());

test('two people and a bot play through a trick on the real Worker', async ({ browser }) => {
  const hostCtx = await browser.newContext();
  const guestCtx = await browser.newContext();
  const host = await hostCtx.newPage();
  const guest = await guestCtx.newPage();
  await launch(host, 'Ala');
  await launch(guest, 'Bartek');

  await host.getByRole('button', { name: '3 players' }).click();
  await expect(host.getByText('Room code')).toBeVisible();
  const code = (await host.locator('.lobby__code b').textContent())!.trim();
  expect(code).toMatch(/^[A-Z0-9]{6}$/);

  await guest.getByLabel('Join with code').fill(code);
  await guest.getByRole('button', { name: 'Join' }).click();
  await expect(host.getByRole('button', { name: /^Seat \d: Bartek$/ })).toBeVisible();

  await host.getByRole('button', { name: 'Add bot' }).click();
  await host.getByRole('button', { name: 'Start game' }).click();
  await expect(host.locator('.zone')).toBeVisible({ timeout: 30_000 });
  await expect(guest.locator('.zone')).toBeVisible({ timeout: 30_000 });

  const noop = async () => {};
  await Promise.all([
    drive(host, { check: noop, stop: afterFirstTrick(host) }),
    drive(guest, { check: noop, stop: afterFirstTrick(guest) }),
  ]);

  // Both humans saw the same completed trick.
  const trickOf = (p: Page) => p.getByLabel('Last trick').getByRole('img').evaluateAll(els => els.map(e => e.getAttribute('aria-label')));
  if (await host.getByLabel('Last trick').isVisible()) expect(await trickOf(guest)).toEqual(await trickOf(host));

  // The debug endpoint documented in CLAUDE.md works and holds the action log.
  const res = await host.request.get(`/api/debug/rooms/${code}`, { headers: { 'x-debug-key': debugKey() } });
  expect(res.status()).toBe(200);
  const dump = (await res.json()) as { meta: { code: string }; log: unknown[]; state: { seq: number } | null };
  expect(dump.meta.code).toBe(code);
  expect(dump.log.length).toBeGreaterThanOrEqual(5);
  expect(dump.state?.seq).toBeGreaterThanOrEqual(5);
  const wrongKey = await host.request.get(`/api/debug/rooms/${code}`, { headers: { 'x-debug-key': 'nope' } });
  expect(wrongKey.status()).toBe(404);

  // History reads the real endpoint: a fresh player has no finished games.
  await host.goto('/history');
  await expect(host.getByText('No finished games yet — play one with friends!')).toBeVisible();

  await hostCtx.close();
  await guestCtx.close();
});

```yaml
name: CI

on:
  push:
    branches: [main, rebuild]
  pull_request:
    branches: [main, rebuild]
  workflow_dispatch:

concurrency:
  group: ci-${{ github.ref }}
  cancel-in-progress: true

env:
  WRANGLER_SEND_METRICS: 'false'

jobs:
  check:
    name: typecheck, unit tests, build
    runs-on: ubuntu-latest
    timeout-minutes: 20
    steps:
      - uses: actions/checkout@v4
      - uses: pnpm/action-setup@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 24
          cache: pnpm
      - run: pnpm install --frozen-lockfile
      - run: pnpm typecheck
      - run: pnpm test
      - name: Build app (initial-JS budget + PWA checks)
        run: pnpm --filter @tysiac/app build

  e2e:
    name: e2e (${{ matrix.suite }})
    needs: check
    runs-on: ubuntu-latest
    timeout-minutes: 45
    strategy:
      fail-fast: false
      matrix:
        suite: [local, worker]
    steps:
      - uses: actions/checkout@v4
      - uses: pnpm/action-setup@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 24
          cache: pnpm
      - run: pnpm install --frozen-lockfile
      - name: Install WebKit
        run: pnpm --filter @tysiac/app exec playwright install --with-deps webkit
      - name: Playwright (vite preview, in-browser rooms, 5 viewports)
        if: matrix.suite == 'local'
        run: pnpm --filter @tysiac/app e2e
      - name: Playwright (wrangler dev, real Worker)
        if: matrix.suite == 'worker'
        run: pnpm --filter @tysiac/app e2e:worker
      - uses: actions/upload-artifact@v4
        if: failure()
        with:
          name: playwright-${{ matrix.suite }}
          path: apps/app/test-results
          retention-days: 7
```
